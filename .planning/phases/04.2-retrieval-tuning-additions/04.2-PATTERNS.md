```ts
  limit?: number;
```

**Row guard** (`toRow`): never trust the RPC payload shape.
```ts
function toRow(raw: unknown): ExactIdRow | null {
  if (typeof raw !== 'object' || raw === null) return null;
  const row = raw as Record<string, unknown>;
  if (typeof row.chunk_id !== 'string' || typeof row.document_id !== 'string') return null;
  const metadata = typeof row.metadata === 'object' && row.metadata !== null && !Array.isArray(row.metadata) ? (row.metadata as Record<string, unknown>) : null;
  return { chunk_id: row.chunk_id, document_id: row.document_id, content: typeof row.content === 'string' ? row.content : '', metadata, matched_field: typeof row.matched_field === 'string' ? row.matched_field : '' };
}
```

**Core RPC call** (`exactIdLookup`): scope assertion first, tenant only from `scopeKey`, values as one array parameter, abort checked before and after, error mapped to `RetrievalBackendError` with the code only.
```ts
export async function exactIdLookup(scope: TenantScope, tokens: ExactIdToken[], options: ExactIdLookupOptions): Promise<ExactIdRow[]> {
  assertTenantScope(scope);
  const values = lookupValues(tokens);
  if (values.length === 0) return [];
  const { client, signal } = options;
  signal.throwIfAborted();
  const response = await client
    .rpc(RPC_NAME, { p_tenant_key: scopeKey(scope).toLowerCase(), p_tokens: values, p_limit: options.limit ?? EXACT_ID_LIMIT })
    .abortSignal(signal);
  signal.throwIfAborted();
  if (response.error) throw new RetrievalBackendError(RPC_NAME, response.error.code ?? null);
  return rpcRows(response.data).map(toRow).filter((row): row is ExactIdRow => row !== null);
}
```

**Stage wrapper** (`exactIdStage`): no input, no call; the count goes to the trace detail, the values never do.
```ts
export async function exactIdStage(query: string, scope: TenantScope, options: ExactIdLookupOptions): Promise<ExactIdStageResult> {
  const tokens = extractExactIds(query);
  if (tokens.length === 0) return { rows: [], tokenCount: 0 };
  return { rows: await exactIdLookup(scope, tokens, options), tokenCount: tokens.length };
}
```

**Do not copy:** `INVOICE_CUE`, `ExactIdKind`, `classifyIdentifier`, `mergeExactCandidates` (legacy path), and the tax-id branches.

## Shared Patterns

### Tenant scope in TypeScript
**Source:** `exactIdLookup` in `src/lib/rag/retrieval/exact-id.ts`
**Apply to:** every new service that calls an RPC (`table-lookup.ts`, `identifiers.ts`, `dedupe/cluster.ts`, `identifiers/patterns.ts`, `tables/rows.ts`, `lexical/resolve-options.ts`)

### Tenant scope and grants in SQL
**Source:** `supabase/migrations/0601_tenant_language_shares.sql`, `0603_semantic_cache.sql`
**Apply to:** every function in 0855..0861
```sql
PERFORM public.rag_assert_tenant_key(p_tenant_key);   -- first statement

REVOKE ALL ON FUNCTION public.<fn>(<full signature>) FROM PUBLIC, anon, authenticated;
GRANT EXECUTE ON FUNCTION public.<fn>(<full signature>) TO service_role;
NOTIFY pgrst, 'reload schema';
```

`SET search_path = ''`, schema-qualified names, `pg_catalog.` built-ins, `OPERATOR(public.<=>)` for pgvector operators, `BEGIN; … COMMIT;`, idempotent (`IF NOT EXISTS`, `CREATE OR REPLACE`, `DROP TRIGGER IF EXISTS`).
